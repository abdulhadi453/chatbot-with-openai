## Frontend Integration

### Quick Start

```typescript
async function sendMessageStreaming(
  conversationId: string,
  content: string,
  onChunk: (text: string) => void
) {
  const response = await fetch(
    `http://localhost:8000/api/conversations/${conversationId}/messages/stream`,
    {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ content })
    }
  );

  const reader = response.body!.getReader();
  const decoder = new TextDecoder();

  while (true) {
    const { done, value } = await reader.read();
    if (done) break;

    const text = decoder.decode(value);
    const lines = text.split('\n');

    for (const line of lines) {
      if (line.startsWith('data: ')) {
        const data = JSON.parse(line.slice(6));
        
        if (data.type === 'chunk') {
          onChunk(data.content); // Update UI immediately
        } else if (data.type === 'done') {
          console.log('Message complete:', data.message);
        }
      }
    }
  }
}
```

### React Component Example

```tsx
import { useState } from 'react';

function ChatMessage({ conversationId }: { conversationId: string }) {
  const [message, setMessage] = useState('');
  const [streaming, setStreaming] = useState('');
  const [isStreaming, setIsStreaming] = useState(false);

  const handleSend = async () => {
    setIsStreaming(true);
    setStreaming('');

    const response = await fetch(
      `http://localhost:8000/api/conversations/${conversationId}/messages/stream`,
      {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ content: message })
      }
    );

    const reader = response.body!.getReader();
    const decoder = new TextDecoder();

    while (true) {
      const { done, value } = await reader.read();
      if (done) {
        setIsStreaming(false);
        break;
      }

      const text = decoder.decode(value);
      const lines = text.split('\n');

      for (const line of lines) {
        if (line.startsWith('data: ')) {
          const data = JSON.parse(line.slice(6));
          
          if (data.type === 'chunk') {
            setStreaming(prev => prev + data.content);
          }
        }
      }
    }
  };

  return (
    <div>
      <input 
        value={message} 
        onChange={(e) => setMessage(e.target.value)}
      />
      <button onClick={handleSend}>Send</button>
      <div className={isStreaming ? 'streaming' : ''}>
        {streaming}
      </div>
    </div>
  );
}
```
